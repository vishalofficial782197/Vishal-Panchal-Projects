function doGet(e) { 
  Logger.log(JSON.stringify(e));
  var result = 'Ok';

  if (e.parameter == 'undefined') {
    result = 'No Parameters';
  } else {
    var sheet_id = '1k1FdavMjKIB9hLvYTBHfXFgmiUc-NaF1kB55ZCSvr4c'; // TODO: put your Spreadsheet ID here
    var sheet = SpreadsheetApp.openById(sheet_id).getActiveSheet();
    
    var newRow = sheet.getLastRow() + 1; 
    var rowData = [];
    
    var Curr_Date = new Date();
    rowData[0] = Curr_Date; // Date in column A
    
    // Time in column B (Asia/Kolkata is correct for India)
    var Curr_Time = Utilities.formatDate(Curr_Date, "Asia/Kolkata", 'HH:mm:ss');
    rowData[1] = Curr_Time; 

    for (var param in e.parameter) {
      Logger.log('In for loop, param=' + param);
      var value = stripQuotes(e.parameter[param]);
      Logger.log(param + ':' + e.parameter[param]);

      switch (param) {
        case 'temperature':
          rowData[2] = value; // Column C
          result = 'Temperature written on column C'; 
          break;

        case 'humidity':
          rowData[3] = value; // Column D
          result += ' ,Humidity written on column D'; 
          break;

        case 'water':
          rowData[4] = value; // Column E (Water Level)
          result += ' ,Water Level written on column E';
          break;

        case 'soil':
          rowData[5] = value; // Column F (Soil Moisture)
          result += ' ,Soil Moisture written on column F';
          break;

        case 'light':
          rowData[6] = value; // Column G (Light)
          result += ' ,Light written on column G';
          break;

        case 'ph':
          rowData[7] = value; // Column H (pH)
          result += ' ,pH written on column H';
          break;

        default:
          // Ignore unsupported parameters instead of overwriting result
          Logger.log("Unsupported parameter: " + param);
      }
    }

    Logger.log(JSON.stringify(rowData));
    var newRange = sheet.getRange(newRow, 1, 1, rowData.length);
    newRange.setValues([rowData]);
  }

  return ContentService.createTextOutput(result);
}

function stripQuotes(value) {
  return value.replace(/^["']|['"]$/g, "");
}
